# Python dependencies for the ECU test framework
#
#   pip install -r requirements.txt
#
# python-can is the standard Python CAN abstraction. It supports every adapter
# this project might meet - SLCAN serial dongles, PEAK PCAN, Kvaser, Vector,
# Linux SocketCAN - behind one API, plus a virtual backend that needs no
# hardware at all.
#
# Versions are pinned to a minimum rather than exactly: the APIs used here are
# stable, and pinning exactly would force the whole project onto one release
# for no benefit in a project this size. A production test bench pins exact
# versions, because a test environment that changes underneath you
# invalidates every result it has ever produced.

python-can>=4.3.0

# Diagnostics (v0.2). Independent open-source implementations of the two
# standards the ECU implements. Using someone else's implementation as the
# tester is the point: agreement with it is evidence of interoperability.
can-isotp>=2.0.0        # ISO 15765-2 transport
udsoncan>=1.23.0        # ISO 14229-1 UDS client

# Optional - only needed on a machine with no C compiler (typically Windows),
